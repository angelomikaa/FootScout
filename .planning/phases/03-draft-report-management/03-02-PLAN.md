```typescript
import { useFetcher } from "react-router";
import type { Report } from "../data/types";

interface DraftBannerProps {
  draft: Report;
  onDiscard: () => void;
  onResume: () => void;
}

export function DraftBanner({ draft, onDiscard, onResume }: DraftBannerProps) {
  const fetcher = useFetcher();
  const isPending = fetcher.state !== "idle";

  const handleDiscard = () => {
    fetcher.submit(
      { intent: "delete-draft", reportId: draft.id },
      { method: "POST", action: "/scout/report" }
    );
    onDiscard();
  };

  const handleResume = () => {
    onResume();
  };

  return (
    <div className="bg-yellow-50 border-l-4 border-yellow-400 p-4 mb-6">
      <div className="flex items-start">
        <div className="flex-shrink-0">
          <svg className="h-5 w-5 text-yellow-400" viewBox="0 0 20 20" fill="currentColor">
            <path fillRule="evenodd" d="M8.257 3.099c.765-1.36 2.722-1.36 3.486 0l5.58 9.92c.75 1.334-.213 2.98-1.742 2.98H4.42c-1.53 0-2.493-1.646-1.743-2.98l5.58-9.92zM11 13a1 1 0 11-2 0 1 1 0 012 0zm-1-8a1 1 0 00-1 1v3a1 1 0 002 0V6a1 1 0 00-1-1z" clipRule="evenodd" />
          </svg>
        </div>
        <div className="ml-3">
          <h3 className="text-sm font-medium text-yellow-800">
            Unsaved Draft
          </h3>
          <p className="mt-1 text-sm text-yellow-700">
            You have an unsaved draft for <span className="font-semibold">{draft.playerName}</span> from {new Date(draft.matchDate).toLocaleDateString()}.

              className="inline-flex items-center px-3 py-1.5 border border-transparent text-xs font-medium rounded shadow-sm text-white bg-yellow-600 hover:bg-yellow-700 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-yellow-500"
```

```typescript
import { useFetcher, useNavigation } from "react-router";
import { DraftBanner } from "./draft-banner";
import type { Report } from "../data/types";

interface ScoutReportFormProps {
  draft: Report | null;
  // ... other props
}

export function ScoutReportForm({ draft, ...props }: ScoutReportFormProps) {
  const fetcher = useFetcher();
  const navigation = useNavigation();
  const [currentStep, setCurrentStep] = useState(draft?.currentStep || 0);
  const [hasResumed, setHasResumed] = useState(false);
  
  // Auto-save on step change
  useEffect(() => {
    if (draft || currentStep > 0) {
      // Only auto-save if we have a draft or have moved from step 0
      const formData = new FormData();
      formData.append("intent", "save-draft");
      formData.append("currentStep", String(currentStep));
      // Add all current form values
      // ...
      fetcher.submit(formData, { method: "POST", action: "/scout/report" });
    }
  }, [currentStep]);
  
  // Draft resume: pre-fill form values
  const defaultValues = useMemo(() => {
    if (draft && !hasResumed) {
      setHasResumed(true);
      return {
        // Map draft fields to form fields
        // ...
      };
    }
    return {};
  }, [draft]);
  
  const form = useForm({ defaultValues });
  
  const handleDiscardDraft = () => {
    setHasResumed(false);
    // Clear form
    form.reset();
  };
  
  const handleResumeDraft = () => {
    // Form already has defaultValues, just ensure step is set
    setCurrentStep(draft?.currentStep || 0);
  };
  
  return (
    <div>
      {draft && !hasResumed && (
        <DraftBanner
          draft={draft}
          onDiscard={handleDiscardDraft}
          onResume={handleResumeDraft}
        />
      )}
      
      {/* Existing form content */}
      {/* ... */}
      
      {/* Draft saved indicator */}
      {fetcher.state === "idle" && fetcher.data?.success && (
        <div className="text-sm text-green-600">Draft saved</div>
      )}
    </div>
  );
}
```
