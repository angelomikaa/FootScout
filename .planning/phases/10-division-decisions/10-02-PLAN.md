**app/routes/division/watchlists.ts** — Create action route (no loader needed):

```typescript
import { type Route } from "./+types/watchlists";
import { json } from "react-router";
import { getDivisionUserIdFromCookie } from "../../cookies.server";
import { createWatchlist, renameWatchlist, deleteWatchlist, addPlayerToWatchlist, removePlayerFromWatchlist } from "../../data/data";

export async function action({ request }: Route.ActionArgs) {
  const formData = await request.formData();
  const action = formData.get("_action") as string;
  const userId = await getDivisionUserIdFromCookie(request);

  if (!userId) return json({ error: "Unauthorized" }, { status: 401 });

  switch (action) {
    case "create": {
      const name = formData.get("name") as string;
      if (!name || name.trim().length === 0) {
        return json({ error: "Name is required" }, { status: 400 });
      }
      const watchlist = await createWatchlist({ name: name.trim(), userId });
      const playerId = formData.get("playerId") as string | null;
      if (playerId) {
        await addPlayerToWatchlist(watchlist.id, playerId);
      }
      return json({ watchlist });
    }
    case "rename": {
      const id = formData.get("id") as string;
      const name = formData.get("name") as string;
      const watchlist = await renameWatchlist(id, name, userId);
      return json({ watchlist });
    }
    case "delete": {
      const id = formData.get("id") as string;
      await deleteWatchlist(id, userId);
      return json({ ok: true });
    }
    case "add": {
      const watchlistId = formData.get("watchlistId") as string;
      const playerId = formData.get("playerId") as string;
      const entry = await addPlayerToWatchlist(watchlistId, playerId);
      return json({ entry });
    }
    case "remove": {
      const watchlistId = formData.get("watchlistId") as string;
      const playerId = formData.get("playerId") as string;
      await removePlayerFromWatchlist(watchlistId, playerId);
      return json({ ok: true });
    }
    default:
      return json({ error: "Invalid action" }, { status: 400 });
  }
}
```

export async function loader({ params, request }: Route.LoaderArgs) {
  const userId = await getDivisionUserIdFromCookie(request);
  const watchlist = await getWatchlistById(params.id);
  if (!watchlist || watchlist.userId !== userId) {

}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Action Route | Untrusted FormData with watchlistId, playerId, name, _action crosses here |
| Action Route → DB | Server-side code executes SQL with user-supplied values |
| Cookie → Identity | division-user-id cookie can be spoofed; watchlist ownership check prevents cross-user access |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-06 | Tampering | watchlists.ts action route | mitigate | Validate `_action` against whitelist ("create", "rename", "delete", "add", "remove"); validate `name` is non-empty string before createWatchlist call |
| T-10-07 | Authorization Bypass | watchlists.$id.tsx loader | mitigate | Loader checks `watchlist.userId !== userId` and throws 404 — prevents viewing another user's watchlist |
| T-10-08 | Tampering | watchlists.$id.ts action route | mitigate | All mutations (remove-player, rename) verify userId ownership via getDivisionUserIdFromCookie + data layer userId check |
| T-10-09 | Information Disclosure | getWatchlistsByUser in players.tsx loader | mitigate | Only watchlists for current userId are fetched; no cross-user data leakage |
| T-10-10 | Denial of Service | watchlist creation (unlimited) | accept | v1 internal tool with cookie identity; rate limiting not required; Supabase migration will add proper auth |
</threat_model>
